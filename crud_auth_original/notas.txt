/CREAR TABLAS/
use mydb;

create table membresia(
id_membresia int primary key auto_increment,
id_usuario int, foreign key (id_usuario) references persona(id),
esta_activa boolean
);

CREATE TABLE facturas (
    id_facturas INT PRIMARY KEY AUTO_INCREMENT,
    fecha_emision DATE NOT NULL,
    importe_pago DECIMAL(10,2) NOT NULL,
    id_plan INT,
    id_usuario INT NOT NULL
);


CREATE TABLE pagos (
    id_pago INT PRIMARY KEY AUTO_INCREMENT,
    id_usuario INT NOT NULL,
    monto DECIMAL(10,2) NOT NULL,
    id_factura INT,
    fecha_pago DATE NOT NULL,
    referencia VARCHAR(100)
);

/////JOIN PERSONAS, PAGOS, FACTURAS//////

SELECT 
    auth_user.id_usuario,
    auth_user.nombre_usuario,
    auth_user.email,
    auth_user.is_staff,
    facturas.id_facturas,
    facturas.fecha_emision,
    facturas.importe_pago,
    facturas.id_plan,
    pagos.id_pago,
    pagos.monto,
    pagos.fecha_pago,
    pagos.referencia
FROM facturas
INNER JOIN auth_user ON facturas.id_usuario = auth_user.id_usuario
INNER JOIN pagos ON facturas.id_facturas = pagos.id_factura AND auth_user.id_usuario = pagos.id_usuario;
